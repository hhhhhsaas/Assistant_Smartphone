def process_user_request(self, user_request: str) -> Dict:
    return self.parse_natural_language(user_request)

def parse_natural_language(self, text: str) -> Dict:
    text_lower = text.lower()
    parsed = {}

    # ─── 1. NHẬN DẠNG NHU CẦU (dòng 88-126) ─────────
    need_map = [
        (['chơi game', 'gaming', 'pubg', 'liên quân'], 'gaming'),
        (['chụp ảnh', 'camera', 'nhiếp ảnh'], 'photography'),
        (['pin trâu', 'pin lâu', 'trâu'], 'battery_life'),
        # ... 14 loại nhu cầu
    ]

    for keywords, need_val in need_map:
        for kw in keywords:
            if kw in text_lower:
                parsed['user_need'] = need_val
                break

    # ─── 2. PARSE NGÂN SÁCH (dòng 130-153) ──────────
    budget_patterns = [
        # "tầm 15 triệu" → ±10%
        (r'(?:tầm|khoảng)\s*(\d+)\s*(?:triệu|tr)?',
         lambda m: (int(m.group(1)) * 1000000 * 0.9,
                   int(m.group(1)) * 1000000 * 1.1)),

        # "từ 10 đến 15 triệu"
        (r'(?:từ)\s*(\d+)\s*(?:triệu|tr)?\s*(?:đến)\s*(\d+)\s*(?:triệu|tr)',
         lambda m: (int(m.group(1)) * 1000000,
                   int(m.group(2)) * 1000000)),
    ]

    for pattern, extractor in budget_patterns:
        match = re.search(pattern, text_lower)
        if match:
            min_b, max_b = extractor(match)
            parsed['budget_min'] = min_b
            parsed['budget'] = max_b
            break

    # ─── 3. NHẬN DẠNG BRAND (dòng 166-182) ──────────
    brand_map = {
        'iphone': 'Apple', 'apple': 'Apple',
        'samsung': 'Samsung', 'galaxy': 'Samsung',
        'xiaomi': 'Xiaomi', 'redmi': 'Xiaomi',
        'oppo': 'OPPO', 'reno': 'OPPO',
    }

    for keyword, brand in brand_map.items():
        if keyword in text_lower:
            parsed['preferred_brand'] = brand
            break

    # ─── 4. DETECT FEATURES (dòng 183-295) ──────────
    # NFC, 5G, sạc không dây, jack tai nghe...
    if 'nfc' in text_lower:
        parsed['required_nfc'] = True
    if '5g' in text_lower:
        parsed['network'] = '5G'

    return parsed

def infer(self, rules: List[Rule], initial_facts: Dict) -> Dict:
    self.memory.set_facts(initial_facts)
    changed = True

    while changed:
        changed = False

        # Lọc rules chưa fire và thỏa điều kiện
        candidates = [
            r for r in rules
            if r.id not in self.memory.activated_rules
            and r.evaluate(self.memory.facts)
        ]

        # Conflict Resolution: sắp xếp theo priority
        ordered = self.strategy.resolve(
            candidates,
            self.memory.activated_rules,
            self.memory.facts
        )

        # Fire rules theo thứ tự
        for rule in ordered:
            self.memory.add_activated_rule(rule.id)

            # Apply actions
            for key, value in rule.actions.items():
                self.memory.add_inferred_fact(key, value)
                self.memory.facts[key] = self.memory.inferred_facts[key]

            changed = True

def add_inferred_fact(self, key: str, value: Any):
    old_value = self.inferred_facts.get(key)

    # Category filter: UNION
    if key == "category_filter":
        old_list = list(old_value) if old_value else []
        new_list = list(value) if isinstance(value, list) else [value]
        merged = list(dict.fromkeys(old_list + new_list))
        self.inferred_facts[key] = merged

    # Min values: MAX
    elif key in ("min_battery", "min_camera_mp"):
        self.inferred_facts[key] = max(float(old_value), float(value))

    # Other keys: override
    else:
        self.inferred_facts[key] = value

def search_by_criteria(self, criteria: Dict, n_results: int = 500):
    results = []
    for fact in self.facts:
        match = True

        # Check brand
        if 'brand' in criteria and fact.get('brand') != criteria['brand']:
            match = False

        # Check price (LOẠI PRICE=0)
        phone_price = fact.get('price', 0)
        if ('min_price' in criteria or 'max_price' in criteria) \
           and (not phone_price or phone_price <= 0):
            match = False  # Loại phones "Liên Hệ"

        if 'min_price' in criteria and phone_price < criteria['min_price']:
            match = False
        if 'max_price' in criteria and phone_price > criteria['max_price']:
            match = False

        # Check category
        if 'category' in criteria:
            cats = criteria['category']
            if fact.get('category') not in cats:
                match = False

        if match:
            results.append(fact)

    return results

def _passes_filters(self, phone: Dict, inferred: Dict, user_input: Dict):
    # Check RAM (dòng 490-499)
    if 'min_ram' in inferred:
        pr = phone.get('ram', '')
        req = int(inferred['min_ram'].replace('GB', ''))
        pv = int(pr.replace('GB', ''))
        if pv < req:
            return False

    # Check Processor Tier (dòng 590-601)
    if 'min_processor_tier' in inferred:
        required_tier = inferred['min_processor_tier']
        processor_str = str(phone.get('processor', '')).lower()

        # Normalize: xóa ® ™
        processor_str = processor_str.replace('®', '').replace('™', '')

        actual_tier = self._get_processor_tier(processor_str)
        tier_order = {'low': 0, 'mid': 1, 'gaming-mid': 1,
                     'upper-mid': 2, 'high': 3}

        if tier_order[actual_tier] < tier_order[required_tier]:
            return False

    # Check Battery (dòng 508-516)
    if 'min_battery' in inferred:
        bt = phone.get('battery', '0')
        bv = int(re.search(r'(\d+)', str(bt)).group(1))
        if bv < inferred['min_battery']:
            return False

    # Check NFC (dòng 528-531)
    if 'required_nfc' in inferred:
        nfc = phone.get('nfc')
        if not nfc or nfc is False:
            return False

    return True

def calculate_score(self, phone_data: Dict, user_preferences: Dict,
                   budget: float = None):
    memberships = self.calculate_memberships(phone_data)

    # Trọng số cho gaming
    weights = {
        'performance': 0.35,
        'screen': 0.30,
        'ram': 0.15,
        'battery': 0.12,
        'price': 0.05,
        'camera': 0.03
    }

    total_score = 0
    total_weight = 0

    for criterion, weight in weights.items():
        if criterion == 'price' and budget > 0:
            # Budget-aware price scoring
            price_fitness = self.calculate_price_budget_fitness(
                phone_data.get('price', 0), budget
            )
            total_score += price_fitness * weight
        else:
            # Fuzzy membership scoring
            max_membership = max(memberships[criterion].values())
            total_score += max_membership * weight

        total_weight += weight

    return total_score / total_weight

def calculate_price_budget_fitness(self, phone_price: float, budget: float):
    ratio = phone_price / budget

    if ratio <= 0.3:  # Quá rẻ
        return 0.3 + (ratio / 0.3) * 0.3
    elif ratio <= 0.5:  # Rẻ, chấp nhận
        return 0.6 + ((ratio - 0.3) / 0.2) * 0.3
    elif ratio <= 1.0:  # Gần budget, tốt nhất
        return 0.9 + ((ratio - 0.5) / 0.5) * 0.1
    elif ratio <= 1.1:  # Hơi vượt
        return 1.0 - ((ratio - 1.0) / 0.1) * 0.3
    else:  # Vượt nhiều
        return max(0.0, 0.7 - (ratio - 1.1) * 2)
